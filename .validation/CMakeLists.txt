cmake_minimum_required(VERSION 3.20)
project(adaptor_assets_validation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(ASSET_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
get_filename_component(REPO_ROOT "${ASSET_ROOT}/../.." ABSOLUTE)
set(CATCH_DIR "${REPO_ROOT}/master_template_fixed/tools/catch")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/main.cpp" "#define CATCH_CONFIG_MAIN\n#include \"catch.hpp\"\n")
add_library(catch_main STATIC "${CMAKE_CURRENT_BINARY_DIR}/main.cpp")
target_include_directories(catch_main PRIVATE "${CATCH_DIR}")
set(WRAPPERS)
foreach(Q RANGE 1 8)
  foreach(I RANGE 1 4)
    set(FILE "${CMAKE_CURRENT_BINARY_DIR}/q${Q}_${I}.cpp")
    file(WRITE "${FILE}" "#define TEST_CASE_INDEX ${I}\n#include \"${ASSET_ROOT}/question_tests/question_${Q}/cases.h\"\n")
    list(APPEND WRAPPERS "${FILE}")
  endforeach()
endforeach()
file(GLOB REFERENCE_SOURCES "${ASSET_ROOT}/reference_solution/src/*.cpp")
add_executable(validate_assets ${WRAPPERS} ${REFERENCE_SOURCES} api_contract.cpp)
find_package(Threads REQUIRED)
target_include_directories(validate_assets PRIVATE "${ASSET_ROOT}/reference_solution/include" "${CATCH_DIR}")
target_precompile_headers(validate_assets PRIVATE "${CATCH_DIR}/catch.hpp")
target_link_libraries(validate_assets PRIVATE catch_main Threads::Threads)
if(MSVC)
  target_compile_options(validate_assets PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(catch_main PRIVATE /utf-8)
else()
  target_compile_options(validate_assets PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME all_assets COMMAND validate_assets)
set_tests_properties(all_assets PROPERTIES TIMEOUT 120)
